\paragraph{Value (embedded FSDS).}
One monitoring stack: batch FSDS on $X$ detects input shift; LOCO--AUUC on a refit-on-REF meta-learner detects ranking decay and localizes feature groups; PO-risk/DRPerm on the same REF/LIVE window detects $Y\mid X$ concept shift when AUUC falls without clear $X$ drift---enabling REALLOCATE-before-RELEARN ops instead of silent uplift degradation.
\emph{Insight:} $\mathrm{AUUC}_{\mathrm{live}}$ and $\Delta_{\mathrm{AUUC}}$ are sensitive to \textbf{both covariate shift and concept drift}; MMD/domain AUC on $X$ and PO-risk separate the failure mode (same FSDS as elsewhere, MSE $\rightarrow$ AUUC in the ranking layer).

\noindent\textbf{Validity:} SRM on uplift treatment $T$ (REF \& LIVE).

\noindent\textbf{FSDS on $X$:}\
$\widehat{\mathrm{MMD}}^2(X_{\mathrm{ref}},X_{\mathrm{live}})$,
$\widehat{\mathrm{AUC}}_{\mathrm{dom}}^{\mathrm{RF}}(X)$,
$\mathrm{ESS}_{\mathrm{ovlp}}(\hat e(W{=}1\mid X))$ (support gate for LOCO).

\noindent\textbf{LOCO--AUUC:}\
meta-learner $\hat\tau$ on REF train;
$\mathrm{AUUC}_{\mathrm{ref}}$, $\mathrm{AUUC}_{\mathrm{live}}$, $\Delta_{\mathrm{AUUC}}$;
group LOCO retrain $\Rightarrow$ $\mathrm{drop}_{\mathrm{live}}^{(g)}$, $\rho=\mathrm{Spearman}(\mathrm{drop}_{\mathrm{ref}},\mathrm{drop}_{\mathrm{live}})$.

\noindent\textbf{FSDS PO-risk on $Y\mid X$:}\
$\tilde Y=(Y-\hat\mu(X))(W-\hat e(X))$,
$\mathrm{PO\text{-}risk}=n^{-1}\sum_i \hat\tau_Y(X_i)^2$,
DRPerm on $W$; PO-LOCO $\mathrm{VIMP}_j$.
