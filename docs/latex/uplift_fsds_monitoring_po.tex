\noindent\textbf{FSDS on $X$:}\ 
$\widehat{\mathrm{MMD}}^2(X_{\mathrm{ref}},X_{\mathrm{live}})$,
$\widehat{\mathrm{AUC}}_{\mathrm{dom}}(X)$,
$\mathrm{ESS}_{\mathrm{ovlp}}(\hat e(W{=}1\mid X))$.

\noindent\textbf{LOCO--AUUC:}\
meta-learner $\hat\tau$ fit on REF train only;
$\mathrm{AUUC}_{\mathrm{ref}}$, $\mathrm{AUUC}_{\mathrm{live}}$, $\Delta_{\mathrm{AUUC}}=\mathrm{AUUC}_{\mathrm{ref}}-\mathrm{AUUC}_{\mathrm{live}}$;
drop group $g$, retrain on REF:
$\mathrm{drop}_{\mathrm{live}}^{(g)}=\mathrm{AUUC}_{\mathrm{live}}^{\mathrm{full}}-\mathrm{AUUC}_{\mathrm{live}}^{(-g)}$;
$\rho=\mathrm{Spearman}(\mathrm{drop}_{\mathrm{ref}},\mathrm{drop}_{\mathrm{live}})$.
